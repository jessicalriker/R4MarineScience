# Calculate monthly mean group size
monthly_summary <- killer_whales %>%
  group_by(Month_A) %>%
  summarise(
    mean_group_size = mean(Group_size, na.rm = TRUE),
    sd_group_size = sd(Group_size, na.rm = TRUE),
    n_group = sum(!is.na(Group_size)),
    se_group_size = sd_group_size / sqrt(n_group)
  )

# Put months in chronological order
monthly_summary <- monthly_summary %>%
  mutate(
    Month_A = factor(
      Month_A,
      levels = month.name
    )
  )

# Plot mean group size by month
ggplot(monthly_summary, aes(x = Month_A, y = mean_group_size)) +
  geom_col(fill = "steelblue") +
   geom_errorbar(
    aes(
      ymin = mean_group_size - se_group_size,
      ymax = mean_group_size + se_group_size
    ),
    width = 0.2
  ) +
  labs(
    x = "Month",
    y = "Mean Group Size",

  )
